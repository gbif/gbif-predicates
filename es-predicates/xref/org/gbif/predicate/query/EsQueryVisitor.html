54  /**
55   * The class translates predicates in GBIF download API to equivalent Elasticsearch query requests.
56   * The json string provided by {@link #buildQuery(Predicate)} can be used with _search get requests
57   * of ES index to produce downloads.
58   */
59  @RequiredArgsConstructor
60  @Slf4j
61  public abstract class EsQueryVisitor<S extends SearchParameter> implements QueryVisitor {
62  
63    private final EsFieldMapper<S> esFieldMapper;
64    private final String defaultChecklistKey;
65  
66    private String getChecklistKey(Predicate predicate) {
67  
68      if (predicate == null) return null;
69  
70      Class<?> clazz = predicate.getClass();
71  
72      try {
73        Method m = clazz.getMethod("getChecklistKey");
74        String checklistKey = (String) m.invoke(predicate);
75        if (checklistKey != null) return checklistKey;
76        return defaultChecklistKey;
77      } catch (Exception e) {
78        return null;
79      }
80    }

82    private String getExactMatchFieldName(IsNotNullPredicate<S> predicate) {
83      if (esFieldMapper.isTaxonomic(predicate.getParameter())) {
84        return esFieldMapper.getChecklistField(getChecklistKey(predicate), predicate.getParameter());
85      }
86  
87      return esFieldMapper.getExactMatchFieldName(predicate.getParameter());
88    }
89  
90    private String getExactMatchFieldName(IsNullPredicate<S> predicate) {
91      if (esFieldMapper.isTaxonomic(predicate.getParameter())) {
92        return esFieldMapper.getChecklistField(getChecklistKey(predicate), predicate.getParameter());
93      }
94  
95      return esFieldMapper.getExactMatchFieldName(predicate.getParameter());
96    }
97  
98    private String getExactMatchOrVerbatimField(SimplePredicate<S> predicate) {
99      if (predicate instanceof EqualsPredicate) {
100       EqualsPredicate<S> equalsPredicate = (EqualsPredicate<S>) predicate;
101       if (esFieldMapper.isTaxonomic(equalsPredicate.getKey())) {
102         return esFieldMapper.getChecklistField(
103             getChecklistKey(equalsPredicate), predicate.getKey());
104       }
105     }
106     if (predicate instanceof LikePredicate) {
107       LikePredicate<S> likePredicate = (LikePredicate<S>) predicate;
108       if (esFieldMapper.isTaxonomic(likePredicate.getKey())) {
109         return esFieldMapper.getChecklistField(getChecklistKey(likePredicate), predicate.getKey());
110       }
111     }
112 
113     return predicate.isMatchCase()
114         ? esFieldMapper.getVerbatimFieldName(predicate.getKey())
115         : esFieldMapper.getExactMatchFieldName(predicate.getKey());
116   }
117 
118   private String getExactMatchOrVerbatimField(InPredicate<S> predicate) {
119     if (esFieldMapper.isTaxonomic(predicate.getKey())) {
120       return esFieldMapper.getChecklistField(getChecklistKey(predicate), predicate.getKey());
121     }
122 
123     return Optional.ofNullable(predicate.isMatchCase()).orElse(Boolean.FALSE)
124         ? esFieldMapper.getVerbatimFieldName(predicate.getKey())
125         : esFieldMapper.getExactMatchFieldName(predicate.getKey());
126   }

128   private String parseParamValue(String value, S parameter) {
129     if (Enum.class.isAssignableFrom(parameter.type())
130         && !Country.class.isAssignableFrom(parameter.type())) {
131       return VocabularyUtils.lookup(value, (Class<Enum<?>>) parameter.type())
132           .map(Enum::name)
133           .orElse(null);
134     }
135     if (Boolean.class.isAssignableFrom(parameter.type())) {
136       return value.toLowerCase();
137     }
138     return value;
139   }

141   /**
142    * Translates a valid {@link org.gbif.api.model.occurrence.Download} object and translates it into
143    * a json query that can be used as the <em>body</em> for _search request of ES index.
144    *
145    * @param predicate to translate
146    * @return body clause
147    */
148   private static final JsonpMapper JSONP_MAPPER = new JacksonJsonpMapper();
149 
150   @Override
151   public String buildQuery(Predicate predicate) throws QueryBuildingException {
152     return getQueryBuilder(predicate)
153         .map(EsQueryVisitor::queryToJson)
154         .orElseGet(EsQueryVisitor::matchAllQueryJson);
155   }
156 
157   private static String queryToJson(Query query) {
158     StringWriter sw = new StringWriter();
159     try (JsonGenerator g = JSONP_MAPPER.jsonProvider().createGenerator(sw)) {
160       query.serialize(g, JSONP_MAPPER);
161     } catch (Exception e) {
162       throw new IllegalStateException(e);
163     }
164     return sw.toString();
165   }
166 
167   private static String matchAllQueryJson() {
168     return queryToJson(Query.of(q -> q.matchAll(m -> m)));
169   }

171   /**
172    * Translates a valid {@link org.gbif.api.model.occurrence.Download} object and translates it into
173    * a json query that can be used as the <em>body</em> for _search request of ES index.
174    *
175    * @param predicate to translate
176    * @return body clause
177    */
178   public Optional<Query> getQueryBuilder(Predicate predicate) throws QueryBuildingException {
179     if (predicate != null) {
180       QueryData queryData = new QueryData();
181       visit(predicate, queryData);
182       return Optional.of(buildRootBoolQuery(queryData));
183     }
184     return Optional.empty();
185   }

187   private static Query buildRootBoolQuery(QueryData queryData) {
188     return buildBoolQuery(
189         queryData.filterQueries, queryData.shouldQueries, queryData.mustNotQueries);
190   }
191 
192   private static Query buildBoolQuery(List<Query> filter, List<Query> should, List<Query> mustNot) {
193     return Query.of(
194         q ->
195             q.bool(
196                 b -> {
197                   if (filter != null && !filter.isEmpty()) b.filter(filter);
198                   if (should != null && !should.isEmpty()) b.should(should);
199                   if (mustNot != null && !mustNot.isEmpty()) b.mustNot(mustNot);
200                   return b;
201                 }));
202   }
203 
204   private static Query termQuery(String field, Object value) {
205     return Query.of(q -> q.term(t -> t.field(field).value(FieldValue.of(value))));
206   }
207 
208   private static Query termsQuery(String field, List<Object> values) {
209     return Query.of(
210         q ->
211             q.terms(
212                 t ->
213                     t.field(field)
214                         .terms(
215                             v ->
216                                 v.value(
217                                     values.stream()
218                                         .map(FieldValue::of)
219                                         .collect(Collectors.toList())))));
220   }
221 
222   private static Query existsQuery(String field) {
223     return Query.of(q -> q.exists(e -> e.field(field)));
224   }
225 
226   private static Query wildcardQuery(String field, String value) {
227     return Query.of(q -> q.wildcard(w -> w.field(field).value(value)));
228   }
229 
230   private static Query nestedQuery(String path, Query inner) {
231     return Query.of(q -> q.nested(n -> n.path(path).query(inner).scoreMode(ChildScoreMode.None)));
232   }

234   /** Ensure value serializes to JSON (e.g. LocalDate -> string) to avoid Jackson errors. */
235   private static Object toJsonSafe(Object value) {
236     if (value == null) return null;
237     if (value instanceof LocalDate) return value.toString();
238     return value;
239   }

241   private static RangeRelation parseRangeRelation(String relation) {
242     if (relation == null) return null;
243     return RangeRelation.valueOf(
244         Character.toUpperCase(relation.charAt(0)) + relation.substring(1).toLowerCase());
245   }

247   private static Query rangeQuery(
248       String field, Object gte, Object lte, Object gt, Object lt, String relation) {
249     RangeRelation rangeRelation = parseRangeRelation(relation);
250     return Query.of(
251         q ->
252             q.range(
253                 r ->
254                     r.untyped(
255                         u -> {
256                           u.field(field);
257                           if (gte != null) u.gte(JsonData.of(toJsonSafe(gte)));
258                           if (lte != null) u.lte(JsonData.of(toJsonSafe(lte)));
259                           if (gt != null) u.gt(JsonData.of(toJsonSafe(gt)));
260                           if (lt != null) u.lt(JsonData.of(toJsonSafe(lt)));
261                           if (rangeRelation != null) u.relation(rangeRelation);
262                           return u;
263                         })));
264   }

266   /**
267    * handle conjunction predicate
268    *
269    * @param predicate conjunction predicate
270    * @param queryData data with the root query builder and the nested path
271    */
272   public void visit(ConjunctionPredicate predicate, QueryData queryData) {
273     Map<String, List<Query>> queriesByNestedPath = new HashMap<>();
274     boolean nonNestedQueriesFound = false;
275 
276     Map<S, List<SimplePredicate<S>>> rangeBoundsByKey =
277         predicate.getPredicates().stream()
278             .filter(
279                 p ->
280                     p instanceof GreaterThanPredicate
281                         || p instanceof GreaterThanOrEqualsPredicate
282                         || p instanceof LessThanPredicate
283                         || p instanceof LessThanOrEqualsPredicate)
284             .map(p -> (SimplePredicate<S>) p)
285             .collect(Collectors.groupingBy(SimplePredicate::getKey));
286 
287     Set<Predicate> handledAsMergedRange = new HashSet<>();
288 
289     for (Map.Entry<S, List<SimplePredicate<S>>> entry : rangeBoundsByKey.entrySet()) {
290       List<SimplePredicate<S>> bounds = entry.getValue();
291       if (bounds.size() < 2) {
292         continue;
293       }
294 
295       S key = entry.getKey();
296       Object gte = null, lte = null, gt = null, lt = null;
297       for (SimplePredicate<S> bound : bounds) {
298         Object value = parseParamValue(bound.getValue(), key);
299         if (bound instanceof GreaterThanPredicate) {
300           gt = value;
301         } else if (bound instanceof GreaterThanOrEqualsPredicate) {
302           gte = value;
303         } else if (bound instanceof LessThanPredicate) {
304           lt = value;
305         } else if (bound instanceof LessThanOrEqualsPredicate) {
306           lte = value;
307         }
308         handledAsMergedRange.add(bound);
309       }
310 
311       Query q =
312           addNullableFieldPredicate(
313               bounds.get(0),
314               rangeQuery(esFieldMapper.getExactMatchFieldName(key), gte, lte, gt, lt, null));
315       if (esFieldMapper.isNestedField(key)) {
316         queryData.rawQueries = List.of(q);
317         queryData.nestedPath = esFieldMapper.getNestedPath(key);
318       } else {
319         queryData.filterQueries.add(q);
320         nonNestedQueriesFound = true;
321       }
322     }
323 
324     for (Predicate subPredicate : predicate.getPredicates()) {
325       if (handledAsMergedRange.contains(subPredicate)) {
326         continue;
327       }
328       try {
329         QueryData mustQueryData = new QueryData();
330         visit(subPredicate, mustQueryData);
331 
332         if (mustQueryData.isNested()) {
333           List<Query> queriesToAdd;
334           if (subPredicate instanceof DisjunctionPredicate && mustQueryData.rawQueries.size() > 1) {
335             Query shouldWrapper =
336                 buildBoolQuery(
337                     Collections.emptyList(), mustQueryData.rawQueries, Collections.emptyList());
338             queriesToAdd = List.of(addNullableFieldPredicate(subPredicate, shouldWrapper));
339           } else {
340             queriesToAdd =
341                 mustQueryData.rawQueries.stream()
342                     .map(rq -> addNullableFieldPredicate(subPredicate, rq))
343                     .collect(Collectors.toList());
344           }
345           queriesByNestedPath
346               .computeIfAbsent(mustQueryData.nestedPath, k -> new ArrayList<>())
347               .addAll(queriesToAdd);
348         } else {
349           Query combined = buildBoolFromQueryData(mustQueryData);
350           queryData.filterQueries.add(addNullableFieldPredicate(subPredicate, combined));
351           nonNestedQueriesFound = true;
352         }
353       } catch (QueryBuildingException ex) {
354         throw new RuntimeException(ex);
355       }
356     }
357 
358     for (Map.Entry<String, List<Query>> entry : queriesByNestedPath.entrySet()) {
359       String key = entry.getKey();
360       List<Query> value = entry.getValue();
361       Query nestedBool = buildBoolQuery(value, Collections.emptyList(), Collections.emptyList());
362       queryData.filterQueries.add(nestedQuery(key, nestedBool));
363       queryData.nestedPath = !nonNestedQueriesFound ? key : null;
364       queryData.rawQueries = value;
365     }
366   }

368   private static Query buildBoolFromQueryData(QueryData d) {
369     return buildBoolQuery(d.filterQueries, d.shouldQueries, d.mustNotQueries);
370   }

372   /**
373    * handle disjunction predicate
374    *
375    * @param predicate disjunction predicate
376    * @param queryData data with the root query builder and the nested path
377    */
378   public void visit(DisjunctionPredicate predicate, QueryData queryData) {
379     Map<S, List<EqualsPredicate<S>>> equalsPredicatesReplaceableByIn = groupEquals(predicate);
380 
381     Map<String, List<Query>> queriesByNestedPath = new HashMap<>();
382     boolean nonNestedQueriesFound = false;
383 
384     for (Predicate subPredicate : predicate.getPredicates()) {
385       try {
386         if (!isReplaceableByInPredicate(subPredicate, equalsPredicatesReplaceableByIn)) {
387           QueryData shouldQueryData = new QueryData();
388           visit(subPredicate, shouldQueryData);
389           Query q =
390               addNullableFieldPredicate(subPredicate, buildBoolFromQueryData(shouldQueryData));
391 
392           if (shouldQueryData.isNested()) {
393             queriesByNestedPath
394                 .computeIfAbsent(shouldQueryData.nestedPath, k -> new ArrayList<>())
395                 .addAll(shouldQueryData.rawQueries);
396           } else {
397             nonNestedQueriesFound = true;
398             queryData.shouldQueries.add(q);
399           }
400 
401           if (subPredicate instanceof SimplePredicate) {
402             equalsPredicatesReplaceableByIn.remove(((SimplePredicate) subPredicate).getKey());
403           }
404         }
405       } catch (QueryBuildingException ex) {
406         throw new RuntimeException(ex);
407       }
408     }
409 
410     for (Map.Entry<String, List<Query>> e : queriesByNestedPath.entrySet()) {
411       String s = e.getKey();
412       List<Query> builders = e.getValue();
413       Query nestedBool = buildBoolQuery(Collections.emptyList(), builders, Collections.emptyList());
414       queryData.shouldQueries.add(nestedQuery(s, nestedBool));
415       queryData.nestedPath = !nonNestedQueriesFound ? s : null;
416       queryData.rawQueries = builders;
417     }
418 
419     if (!equalsPredicatesReplaceableByIn.isEmpty()) {
420       Optional<S> geoTimeParam = getParam(OccurrenceSearchParameter.GEOLOGICAL_TIME.name());
421       if (geoTimeParam.isPresent()
422           && equalsPredicatesReplaceableByIn.containsKey(geoTimeParam.get())) {
423         List<Query> queryBuilders = new ArrayList<>();
424         equalsPredicatesReplaceableByIn
425             .get(geoTimeParam.get())
426             .forEach(
427                 ep ->
428                     queryBuilders.add(
429                         termQuery(
430                             getExactMatchOrVerbatimField(ep),
431                             parseParamValue(ep.getValue(), ep.getKey()))));
432 
433         if (!queryBuilders.isEmpty()) {
434           nonNestedQueriesFound = true;
435           queryData.shouldQueries.addAll(queryBuilders);
436         }
437         equalsPredicatesReplaceableByIn.remove(geoTimeParam.get());
438       }
439 
440       if (!equalsPredicatesReplaceableByIn.isEmpty()) {
441         Map<String, List<Query>> replaceableQueriesByNestedPath = new HashMap<>();
442         for (InPredicate<S> ep : toInPredicates(equalsPredicatesReplaceableByIn)) {
443           Query termsQ =
444               termsQuery(
445                   getExactMatchOrVerbatimField(ep),
446                   ep.getValues().stream()
447                       .map(v -> parseParamValue(v, ep.getKey()))
448                       .collect(Collectors.toList()));
449 
450           if (esFieldMapper.isNestedField(ep.getKey())) {
451             replaceableQueriesByNestedPath
452                 .computeIfAbsent(esFieldMapper.getNestedPath(ep.getKey()), k -> new ArrayList<>())
453                 .add(termsQ);
454           } else {
455             queryData.shouldQueries.add(termsQ);
456             nonNestedQueriesFound = true;
457           }
458         }
459 
460         for (Map.Entry<String, List<Query>> entry : replaceableQueriesByNestedPath.entrySet()) {
461           String key = entry.getKey();
462           List<Query> value = entry.getValue();
463           Query nestedBool =
464               buildBoolQuery(Collections.emptyList(), value, Collections.emptyList());
465           queryData.shouldQueries.add(nestedQuery(key, nestedBool));
466           queryData.nestedPath = !nonNestedQueriesFound ? key : null;
467           queryData.rawQueries = value;
468         }
469       }
470     }
471   }

473   /** Checks if a predicate has in grouped and can be replaced later by a InPredicate. */
474   private boolean isReplaceableByInPredicate(
475       Predicate predicate, Map<S, List<EqualsPredicate<S>>> equalsPredicatesReplaceableByIn) {
476     if (!equalsPredicatesReplaceableByIn.isEmpty() && predicate instanceof EqualsPredicate) {
477       EqualsPredicate<S> equalsPredicate = (EqualsPredicate<S>) predicate;
478 
479       // check it is not a range predicate e.g. YEAR 2000,*
480       if (SearchTypeValidator.isNumericRange(equalsPredicate.getValue())
481           || SearchTypeValidator.isDateRange(equalsPredicate.getValue())) {
482         return false;
483       }
484 
485       return equalsPredicatesReplaceableByIn.containsKey(equalsPredicate.getKey())
486           && equalsPredicatesReplaceableByIn
487               .get(equalsPredicate.getKey())
488               .contains(equalsPredicate);
489     }
490     return false;
491   }

493   /** Groups all equals predicates by search parameter. */
494   private Map<S, List<EqualsPredicate<S>>> groupEquals(DisjunctionPredicate predicate) {
495     return predicate.getPredicates().stream()
496         .filter(p -> p instanceof EqualsPredicate)
497         .filter(p -> ((EqualsPredicate<?>) p).getChecklistKey() == null)
498         .map(p -> (EqualsPredicate<S>) p)
499         .collect(Collectors.groupingBy(EqualsPredicate::getKey))
500         .entrySet()
501         .stream()
502         .filter(e -> e.getValue().size() > 1)
503         .collect(Collectors.toMap(Map.Entry::getKey, Map.Entry::getValue));
504   }
505 
506   /** Transforms the grouped EqualsPredicates into InPredicates. */
507   private List<InPredicate<S>> toInPredicates(Map<S, List<EqualsPredicate<S>>> equalPredicates) {
508     return equalPredicates.entrySet().stream()
509         .map(
510             e ->
511                 e.getValue().stream()
512                     .collect(Collectors.groupingBy(EqualsPredicate::isMatchCase))
513                     .entrySet()
514                     .stream()
515                     .map(
516                         group ->
517                             new InPredicate<S>(
518                                 e.getKey(),
519                                 group.getValue().stream()
520                                     .map(EqualsPredicate::getValue)
521                                     .collect(Collectors.toSet()),
522                                 group.getKey()))
523                     .collect(Collectors.toList()))
524         .flatMap(List::stream)
525         .collect(Collectors.toList());
526   }

528   /**
529    * handles EqualPredicate
530    *
531    * @param predicate equalPredicate
532    */
533   public void visit(EqualsPredicate<S> predicate, QueryData queryData) {
534     S parameter = predicate.getKey();
535     String field = esFieldMapper.getExactMatchFieldName(parameter);
536     Query q;
537 
538     if ((Number.class.isAssignableFrom(predicate.getKey().type())
539             || paramEquals(predicate.getKey(), OccurrenceSearchParameter.GEOLOGICAL_TIME.name()))
540         && SearchTypeValidator.isNumericRange(predicate.getValue())) {
541       Range<Double> decimalRange = SearchTypeValidator.parseDecimalRange(predicate.getValue());
542       Double gte = decimalRange.hasLowerBound() ? decimalRange.lowerEndpoint() : null;
543       Double lte = decimalRange.hasUpperBound() ? decimalRange.upperEndpoint() : null;
544       String rel =
545           paramEquals(predicate.getKey(), OccurrenceSearchParameter.GEOLOGICAL_TIME.name())
546               ? "within"
547               : null;
548       q = addNullableFieldPredicate(predicate, rangeQuery(field, gte, lte, null, null, rel));
549     } else if (Date.class.isAssignableFrom(predicate.getKey().type())
550         && SearchTypeValidator.isDateRange(predicate.getValue())) {
551       Range<LocalDate> dateRange = IsoDateParsingUtils.parseDateRange(predicate.getValue());
552       Object gte = dateRange.hasLowerBound() ? dateRange.lowerEndpoint().toString() : null;
553       Object lt = dateRange.hasUpperBound() ? dateRange.upperEndpoint().toString() : null;
554       q = addNullableFieldPredicate(predicate, rangeQuery(field, gte, null, null, lt, "within"));
555     } else if (IsoDateInterval.class.isAssignableFrom(predicate.getKey().type())) {
556       Range<LocalDate> dateRange = IsoDateParsingUtils.parseDateRange(predicate.getValue());
557       Object gte = dateRange.hasLowerBound() ? dateRange.lowerEndpoint().toString() : null;
558       Object lt = dateRange.hasUpperBound() ? dateRange.upperEndpoint().toString() : null;
559       q = addNullableFieldPredicate(predicate, rangeQuery(field, gte, null, null, lt, "within"));
560     } else {
561       q =
562           addNullableFieldPredicate(
563               predicate,
564               termQuery(
565                   getExactMatchOrVerbatimField(predicate),
566                   parseParamValue(predicate.getValue(), parameter)));
567     }
568 
569     addFilterQuery(q, queryData, parameter);
570   }

572   private final Function<String, Object> parseDate =
573       d -> "*".equals(d) ? null : IsoDateParsingUtils.parseDate(d);
574 
575   private final Function<String, Object> parseInteger =
576       d -> "*".equals(d) ? null : Integer.parseInt(d);
577 
578   private final Function<String, Object> parseDouble =
579       d -> "*".equals(d) ? null : Double.parseDouble(d);

581   public void visit(RangePredicate<S> predicate, QueryData queryData)
582       throws QueryBuildingException {
583     String field = esFieldMapper.getExactMatchFieldName(predicate.getKey());
584     Object gte = null, lte = null, gt = null, lt = null;
585     Function<String, Object> initialiser;
586     if (Integer.class.isAssignableFrom(predicate.getKey().type())) {
587       initialiser = parseInteger;
588     } else if (Double.class.isAssignableFrom(predicate.getKey().type())) {
589       initialiser = parseDouble;
590     } else if (Date.class.isAssignableFrom(predicate.getKey().type())) {
591       initialiser = parseDate;
592     } else {
593       addFilterQuery(Query.of(q -> q.matchAll(m -> m)), queryData, predicate.getKey());
594       return;
595     }
596     if (predicate.getValue().getGte() != null)
597       gte = initialiser.apply(predicate.getValue().getGte());
598     if (predicate.getValue().getLte() != null)
599       lte = initialiser.apply(predicate.getValue().getLte());
600     if (predicate.getValue().getGt() != null) gt = initialiser.apply(predicate.getValue().getGt());
601     if (predicate.getValue().getLt() != null) lt = initialiser.apply(predicate.getValue().getLt());
602     Query rqb = rangeQuery(field, gte, lte, gt, lt, null);
603     addFilterQuery(rqb, queryData, predicate.getKey());
604   }

660   /**
661    * handle IN Predicate
662    *
663    * @param predicate InPredicate
664    * @param queryData data with the root query builder and the nested path
665    */
666   public void visit(InPredicate<S> predicate, QueryData queryData) {
667     S parameter = predicate.getKey();
668 
669     if (paramEquals(parameter, OccurrenceSearchParameter.EVENT_DATE.name())) {
670       predicate
671           .getValues()
672           .forEach(
673               value -> {
674                 try {
675                   QueryData shouldQueryData = new QueryData();
676                   Predicate subPredicate =
677                       new EqualsPredicate<>(
678                           getParam(OccurrenceSearchParameter.EVENT_DATE.name()).get(),
679                           value,
680                           predicate.isMatchCase(),
681                           getChecklistKey(predicate));
682                   visit(subPredicate, shouldQueryData);
683                   queryData.shouldQueries.add(
684                       addNullableFieldPredicate(
685                           subPredicate, buildBoolFromQueryData(shouldQueryData)));
686                 } catch (QueryBuildingException ex) {
687                   throw new RuntimeException(ex);
688                 }
689               });
690     } else if (paramEquals(parameter, OccurrenceSearchParameter.GEOLOGICAL_TIME.name())) {
691       List<Predicate> allPredicates =
692           predicate.getValues().stream()
693               .map(v -> new EqualsPredicate<>(predicate.getKey(), v, false))
694               .collect(Collectors.toList());
695       visit(new DisjunctionPredicate(allPredicates), queryData);
696     } else {
697       Query termsQ =
698           termsQuery(
699               getExactMatchOrVerbatimField(predicate),
700               predicate.getValues().stream()
701                   .map(v -> parseParamValue(v, parameter))
702                   .collect(Collectors.toList()));
703       addFilterQuery(termsQ, queryData, predicate.getKey());
704     }
705   }

751   private void addNotPredicateNonNullFilters(Predicate predicate, QueryData queryData) {
752     Map<S, Set<String>> fieldsByParameter = new LinkedHashMap<>();
753     collectNotPredicateFields(predicate, fieldsByParameter);
754 
755     fieldsByParameter.forEach(
756         (parameter, fields) ->
757             fields.forEach(field -> addFilterQuery(existsQuery(field), queryData, parameter)));
758   }

760   @SuppressWarnings("unchecked")
761   private void collectNotPredicateFields(
762       Predicate predicate, Map<S, Set<String>> fieldsByParameter) {
763     if (predicate instanceof ConjunctionPredicate) {
764       ((ConjunctionPredicate) predicate)
765           .getPredicates()
766           .forEach(p -> collectNotPredicateFields(p, fieldsByParameter));
767     } else if (predicate instanceof DisjunctionPredicate) {
768       ((DisjunctionPredicate) predicate)
769           .getPredicates()
770           .forEach(p -> collectNotPredicateFields(p, fieldsByParameter));
771     } else if (predicate instanceof InPredicate) {
772       InPredicate<S> p = (InPredicate<S>) predicate;
773       addNotPredicateFieldToNonNullChecks(
774           fieldsByParameter, p.getKey(), getExactMatchOrVerbatimField(p));
775     } else if (predicate instanceof SimplePredicate) {
776       SimplePredicate<S> p = (SimplePredicate<S>) predicate;
777       addNotPredicateFieldToNonNullChecks(
778           fieldsByParameter, p.getKey(), getExactMatchOrVerbatimField(p));
779     }
780   }

783       Map<S, Set<String>> fieldsByParameter, S parameter, String fieldName) {
784     fieldsByParameter.computeIfAbsent(parameter, ignored -> new LinkedHashSet<>()).add(fieldName);

798   /** Builds a geo_shape query (WITHIN relation) from WKT geometry. */
799   public Query buildGeoShapeQuery(String wkt) {
800     Geometry geometry;
801     try {
802       geometry = new WKTReader().read(wkt);
803     } catch (ParseException e) {
804       throw new IllegalArgumentException(e.getMessage(), e);
805     }
806 
807     String type =
808         "LinearRing".equals(geometry.getGeometryType())
809             ? "LINESTRING"
810             : geometry.getGeometryType().toUpperCase();
811 
812     Object geoJson;
813     if (("POINT").equals(type)) {
814       Coordinate c = geometry.getCoordinate();
815       geoJson = Map.of("type", "Point", "coordinates", List.of(c.x, c.y));
816     } else if ("LINESTRING".equals(type)) {
817       geoJson =
818           Map.of(
819               "type",
820               "LineString",
821               "coordinates",
822               Arrays.stream(geometry.getCoordinates())
823                   .map(c -> List.of(c.x, c.y))
824                   .collect(Collectors.toList()));
825     } else if ("POLYGON".equals(type)) {
826       Polygon poly = (Polygon) geometry;
827       List<List<List<Double>>> rings = new ArrayList<>();
828       rings.add(
829           Arrays.stream(normalizePolygonCoordinates(poly.getExteriorRing().getCoordinates()))
830               .map(c -> List.<Double>of(c.x, c.y))
831               .collect(Collectors.toList()));
832       for (int i = 0; i < poly.getNumInteriorRing(); i++) {
833         rings.add(
834             Arrays.stream(normalizePolygonCoordinates(poly.getInteriorRingN(i).getCoordinates()))
835                 .map(c -> List.<Double>of(c.x, c.y))
836                 .collect(Collectors.toList()));
837       }
838       geoJson = Map.of("type", "Polygon", "coordinates", rings);
839     } else if ("MULTIPOLYGON".equals(type)) {
840       List<List<List<List<Double>>>> polygons = new ArrayList<>();
841       for (int i = 0; i < geometry.getNumGeometries(); i++) {
842         Polygon poly = (Polygon) geometry.getGeometryN(i);
843         List<List<List<Double>>> rings = new ArrayList<>();
844         rings.add(
845             Arrays.stream(normalizePolygonCoordinates(poly.getExteriorRing().getCoordinates()))
846                 .map(c -> List.<Double>of(c.x, c.y))
847                 .collect(Collectors.toList()));
848         for (int h = 0; h < poly.getNumInteriorRing(); h++) {
849           rings.add(
850               Arrays.stream(normalizePolygonCoordinates(poly.getInteriorRingN(h).getCoordinates()))
851                   .map(c -> List.<Double>of(c.x, c.y))
852                   .collect(Collectors.toList()));
853         }
854         polygons.add(rings);
855       }
856       geoJson = Map.of("type", "MultiPolygon", "coordinates", polygons);
857     } else {
858       throw new IllegalArgumentException(type + " shape is not supported");
859     }
860 
861     GeoShapeFieldQuery shapeQuery =
862         GeoShapeFieldQuery.of(s -> s.shape(JsonData.of(geoJson)).relation(GeoShapeRelation.Within));
863     return Query.of(
864         q -> q.geoShape(g -> g.field(esFieldMapper.getGeoShapeField()).shape(shapeQuery)));
865   }

867   /** Eliminates consecutive duplicates. The order is preserved. */
868   static Coordinate[] normalizePolygonCoordinates(Coordinate[] coordinates) {
869     List<Coordinate> normalizedCoordinates = new ArrayList<>();
870 
871     // we always have to keep the fist and last coordinates
872     int i = 0;
873     normalizedCoordinates.add(i++, coordinates[0]);
874 
875     for (int j = 1; j < coordinates.length; j++) {
876       if (!coordinates[j - 1].equals(coordinates[j])) {
877         normalizedCoordinates.add(i++, coordinates[j]);
878       }
879     }
880 
881     return normalizedCoordinates.toArray(new Coordinate[0]);
882   }

919   private void visit(Object object, QueryData queryData) throws QueryBuildingException {
920     Method method;
921     try {
922       method = getClass().getMethod("visit", object.getClass(), QueryData.class);
923     } catch (NoSuchMethodException e) {
924       log.warn(
925           "Visit method could not be found. That means a unknown Predicate has been passed", e);
926       throw new IllegalArgumentException("Unknown Predicate", e);
927     }
928     try {
929       method.invoke(this, object, queryData);
930     } catch (IllegalAccessException e) {
931       log.error(
932           "This error shouldn't occur if all visit methods are public. Probably a programming error",
933           e);
934       throw new RuntimeException(e);
935     } catch (InvocationTargetException e) {
936       log.info("Exception thrown while building the query", e);
937       throw new QueryBuildingException(e);
938     }
939   }

941   private void addFilterQuery(Query q, QueryData queryData, S searchParameter) {
942     if (esFieldMapper.isNestedField(searchParameter)) {
943       queryData.filterQueries.add(nestedQuery(esFieldMapper.getNestedPath(searchParameter), q));
944       queryData.rawQueries = List.of(q);
945       queryData.nestedPath = esFieldMapper.getNestedPath(searchParameter);
946     } else {
947       queryData.filterQueries.add(q);
948     }
949   }
